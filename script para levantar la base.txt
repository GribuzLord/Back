
CREATE SCHEMA `gestion_ets_escom` DEFAULT CHARACTER SET utf8mb4;
 use gestion_ets_escom;


-- ============= CREACION DE TABLAS =============
-- 1. Tabla de Carreras (Catálogo)
CREATE TABLE carreras (
    id_carrera INT AUTO_INCREMENT PRIMARY KEY,
    nombre_carrera VARCHAR(100) NOT NULL
);

-- 2. Tabla de Unidades de Aprendizaje (Materias)
CREATE TABLE materias (
    id_materia INT AUTO_INCREMENT PRIMARY KEY,
    nombre_materia VARCHAR(150) NOT NULL,
    semestre INT NOT NULL,
    id_carrera INT,
    FOREIGN KEY (id_carrera) REFERENCES carreras(id_carrera) ON DELETE CASCADE
);

-- 3. Tabla de Edificios y Salones [cite: 15]
CREATE TABLE salones (
    id_salon INT AUTO_INCREMENT PRIMARY KEY,
    edificio VARCHAR(50) NOT NULL,
    numero_salon VARCHAR(20) NOT NULL
);

-- 4. Tabla de Profesores (Para el Módulo Público) [cite: 9]
CREATE TABLE profesores (
    id_profesor INT AUTO_INCREMENT PRIMARY KEY,
    nombre_completo VARCHAR(150) NOT NULL
);

-- 5. Tabla Principal de Exámenes ETS [cite: 4, 9]
CREATE TABLE examenes_ets (
    id_ets INT AUTO_INCREMENT PRIMARY KEY,
    id_materia INT,
    fecha DATE NOT NULL,
    turno ENUM('Matutino', 'Vespertino') NOT NULL,
    id_salon INT,
    id_profesor INT,
    FOREIGN KEY (id_materia) REFERENCES materias(id_materia),
    FOREIGN KEY (id_salon) REFERENCES salones(id_salon),
    FOREIGN KEY (id_profesor) REFERENCES profesores(id_profesor)
);

-- 6. Tabla de Usuarios (Módulo Administrativo) [cite: 11, 12]
CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL, -- Aquí guardarás el hash de bcrypt
    rol ENUM('admin', 'consultor') DEFAULT 'admin'
);
